cmake_minimum_required(VERSION 3.21)
project(RuntimeReservationContract LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This contract exercises Windows sharing reservations")
endif()
set(AYTHER_RUNTIME_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../AYTHER-Runtime" CACHE PATH "Runtime source checkout")
if(NOT EXISTS "${AYTHER_RUNTIME_SOURCE}/src/file_reservation.h")
  message(FATAL_ERROR "Set AYTHER_RUNTIME_SOURCE to the Runtime checkout with file_reservation.h")
endif()
add_executable(reservation_contract reservation_contract.cpp)
target_compile_features(reservation_contract PRIVATE cxx_std_20)
target_compile_options(reservation_contract PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/EHsc>)
target_include_directories(reservation_contract PRIVATE "${AYTHER_RUNTIME_SOURCE}/src")
enable_testing()
add_test(NAME runtime_reservation_contract COMMAND reservation_contract)
